cmake_minimum_required(VERSION 3.8)
project(audition_behaviors VERSION 0.1.0)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(audition_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(geometry_msgs REQUIRED)

add_executable(attention_manager src/attention_manager.cpp)
target_compile_features(attention_manager PUBLIC cxx_std_17)
target_include_directories(attention_manager PRIVATE include)
ament_target_dependencies(attention_manager rclcpp audition_msgs)
add_executable(attention_tf_bridge src/attention_tf_bridge.cpp)
target_compile_features(attention_tf_bridge PUBLIC cxx_std_17)
target_include_directories(attention_tf_bridge PRIVATE include)
ament_target_dependencies(attention_tf_bridge rclcpp audition_msgs tf2 tf2_ros geometry_msgs)
install(TARGETS attention_manager attention_tf_bridge DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(attention_tf_expiry_test test/test_attention_tf_expiry.cpp)
  target_compile_features(attention_tf_expiry_test PUBLIC cxx_std_17)
  target_include_directories(attention_tf_expiry_test PRIVATE include)
  ament_add_gtest(attention_tf_geometry_test test/test_attention_tf_geometry.cpp)
  target_compile_features(attention_tf_geometry_test PUBLIC cxx_std_17)
  target_include_directories(attention_tf_geometry_test PRIVATE include)
  ament_target_dependencies(attention_tf_geometry_test audition_msgs geometry_msgs)
  ament_add_gtest(attention_policy_test test/test_attention_policy.cpp)
  target_compile_features(attention_policy_test PUBLIC cxx_std_17)
  target_include_directories(attention_policy_test PRIVATE include)
  ament_target_dependencies(attention_policy_test audition_msgs)
  ament_add_gtest(attention_publication_gate_test test/test_attention_publication_gate.cpp)
  target_compile_features(attention_publication_gate_test PUBLIC cxx_std_17)
  target_include_directories(attention_publication_gate_test PRIVATE include)
endif()
ament_package()
